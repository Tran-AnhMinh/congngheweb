class GioHang {
	constructor() {
		this.dsDong = [];
	}

	them(sanPham, soLuong) {
		if (typeof soLuong !== "number" || !Number.isInteger(soLuong) || soLuong <= 0) {
			return false;
		}
		const dong = this.dsDong.find(d => d.id === sanPham.id);
		if (dong) {
			dong.soLuong += soLuong;
		} else {
			this.dsDong.push({
				id: sanPham.id,
				ten: sanPham.ten,
				loai: sanPham.loai,
				gia: sanPham.gia,
				soLuong: soLuong
			});
		}
		return true;
	}

	xoa(id) {
		const index = this.dsDong.findIndex(d => d.id === id);
		if (index !== -1) {
			this.dsDong.splice(index, 1);
			return true;
		}
		return false;
	}

	tongTien() {
		return this.dsDong.reduce((tong, d) => tong + d.gia * d.soLuong, 0);
	}

	thongKeTheoLoai() {
		const thongKe = {};
		for (const d of this.dsDong) {
			thongKe[d.loai] = (thongKe[d.loai] || 0) + d.gia * d.soLuong;
		}
		return thongKe;
	}

	dongDatNhat() {
		if (this.dsDong.length === 0) {
			return null;
		}
		let maxDong = this.dsDong[0];
		let maxTien = maxDong.gia * maxDong.soLuong;
		for (let i = 1; i < this.dsDong.length; i++) {
			const tien = this.dsDong[i].gia * this.dsDong[i].soLuong;
			if (tien > maxTien) {
				maxTien = tien;
				maxDong = this.dsDong[i];
			}
		}
		return {
			id: maxDong.id,
			ten: maxDong.ten,
			loai: maxDong.loai,
			gia: maxDong.gia,
			soLuong: maxDong.soLuong,
			thanhTien: maxTien
		};
	}

	layDanhSach() {
		return this.dsDong.map(d => ({
			id: d.id,
			ten: d.ten,
			loai: d.loai,
			gia: d.gia,
			soLuong: d.soLuong,
			thanhTien: d.gia * d.soLuong
		}));
	}

	xuatJSON() {
		return JSON.stringify(this.dsDong);
	}
}

function napGioHang(chuoiJSON) {
	const gio = new GioHang();
	try {
		const mang = JSON.parse(chuoiJSON);
		if (Array.isArray(mang)) {
			gio.dsDong = mang.map(d => ({
				id: d.id,
				ten: d.ten,
				loai: d.loai,
				gia: d.gia,
				soLuong: d.soLuong
			}));
		}
	} catch (e) {
	}
	return gio;
}

function capNhatLoc() {
	const tuKhoa = oTimKiem.value.toLowerCase();
	const cacMuc = Array.from(ds.querySelectorAll("li"));
	let soHien = 0;

	for (const muc of cacMuc) {
		const vanBan = (muc.querySelector("h3").textContent + " " + muc.querySelector("p").textContent).toLowerCase();
		if (tuKhoa === "" || vanBan.includes(tuKhoa)) {
			muc.classList.remove("an");
			soHien++;
		} else {
			muc.classList.add("an");
		}
	}

	if (soHien === 0) {
		khongThay.classList.remove("an");
		khongThay.textContent = `Không tìm thấy tin nào cho "${oTimKiem.value}"`;
	} else {
		khongThay.classList.add("an");
	}

	boDem.textContent = `Đang hiển thị ${soHien}/${cacMuc.length} tin`;
}

oTimKiem.addEventListener("keyup", capNhatLoc);
capNhatLoc();
